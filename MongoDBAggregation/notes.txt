//ANCHOR - Question - 4 : Which country has the highest number of required users ? 

{
    $group: {
      _id: "$company.location.country",
      count:{
        $sum: 1
      }
    }
  },
  {
    $sort: {
      count:-1
    }
  },{
    $limit: 1
}


Output:-


{
  "_id": "Germany",
  "count": 261
}

//ANCHOR - what is average number of tags per user ? 

[

{
    $unwind: {
      path: "$tags",
    }
  },{
    $group: {
      _id: "$_id",
      count:{
        $sum : 1 ,
      }
    }
  },
  {
    $group: {
      _id: null,
      average:{
        $avg : "$count"
      }
    }
  }
]


Output:- 

{
  "_id": null,
  "average": 3.556
}

             │

    	average:{

  }

      averageNumberofTags: {
        $avg: "$numberOfTags"
      }
    }

//ANCHOR - Question:15:- lookup in authorstable 



[{
  $lookup: {
    from: "authors",
    localField: "author_id",
    foreignField: "_id",
    as: "author_details"
  }
},
 {
   $addFields: {
     author_details: {
       $first : "$author_details"
     }
   }
 }

]

Using ---- >  $arrayElemAt


[{
  $lookup: {
    from: "authors",
    localField: "author_id",
    foreignField: "_id",
    as: "author_details"
  }
},
 {
   $addFields: {
     author_details: {
     	$arrayElemAt : ["$author_details",0] -----> 1st parameter is array on which we wanna do operation and second one is index on which index in the array
     }
   }
 }
